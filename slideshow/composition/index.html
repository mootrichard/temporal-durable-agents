src/temporal/workflows.ts
export async function FixWorkflow(input) {
  const planTurn = await activities.runCodexTurn(...);
  const [source, tests, initialTests] = await Promise.all([investigate(...), investigate(...), activities.runTests(...)]);
  const implementation = await activities.runCodexTurn(...);
  await activities.runTests(...);
  const diff = await activities.getDiff(input.workspace);
  return emit({ type: 'complete', summary, diff });
}
