## !!steps Start a Workflow before the Worker

!change Process start → Workflow start

The original path starts one process. The Temporal path creates a stable
Workflow ID, then starts a Worker for that Workflow.

```ts !before src/supervisor/fleet-supervisor.ts · original process
// The supervisor starts the coordinator process directly.
// !focus(2:6)
this.spawnFleet(managed);

private spawnFleet(managed: ManagedRun): void {
  const entrypoint = 'src/baseline/process.ts';
  const child = spawn(process.execPath, ['--import', 'tsx', entrypoint], {
    detached: true,
    env: { ...process.env, DEMO_RUN_ID: managed.runId, DEMO_WORKSPACE: managed.workspace },
  });
}
```

## !!steps Move the coordinator sequence into a Workflow

!change Orchestrator method → Workflow function

The sequence stays in TypeScript. Temporal records each scheduled command and
completed result in Event History.

```ts !before src/baseline/orchestrator.ts · original
async run(
  input: BaselineRunInput,
  onSnapshot: (snapshot: RunSnapshot) => void = () => undefined,
): Promise<RunSnapshot> {
  let snapshot = createInitialSnapshot(
    input.runId, 'baseline', input.runnerMode,
  );
  const emit = (event: RunEvent) => {
    snapshot = applyRunEvent(snapshot, event);
    onSnapshot(structuredClone(snapshot));
    return snapshot;
  };

  // !focus(1:6)
  const planTurn = await runCodex('coordinator', {
    role: 'planner',
    prompt: plannerPrompt,
    sandboxMode: 'read-only',
    outputSchema: delegationPlanJsonSchema,
  });
}
```

```ts !after src/temporal/workflows.ts · with Temporal
// !focus(1:4)
export async function FixWorkflow(input: FixWorkflowInput) {
  let snapshot = createInitialSnapshot(
    input.runId, 'temporal', input.runnerMode,
  );
  const emit = (event: RunEvent) => {
    snapshot = applyRunEvent(snapshot, event);
    return snapshot;
  };
  setHandler(snapshotQuery, () => snapshot);

  // !focus(1:7)
  const planTurn = await activities.runCodexTurn({
    role: 'planner',
    prompt: plannerPrompt,
    workspace: input.workspace,
    sandboxMode: 'read-only',
    runnerMode: input.runnerMode,
    outputSchema: delegationPlanJsonSchema,
  });
}
```
